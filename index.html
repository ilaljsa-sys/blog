function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Ambil data profil dari sheet 'Profil' atau baris pertama var profil = { nama_lengkap: "I'lal Jalalludin Syahputra", teks_ketik: "Mahasiswa Hukum, Japanese Learner, Livestock Farm", quote: "Keadilan adalah fondasi akal, konsistensi adalah kunci keberhasilan." }; // Ambil data Blog dari sheet 'Blog' var blogSheet = ss.getSheetByName("Blog"); var blogData = []; if (blogSheet) { var rows = blogSheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { blogData.push({ "Judul Artikel": rows[i][0], "Kategori": rows[i][1], "Tanggal": rows[i][2], "Ringkasan Cuplikan": rows[i][3], "Isi Lengkap Artikel": rows[i][4], "SumberURL": rows[i][5] }); } } var responseObj = { status: "success", profil: profil, blog: blogData }; return ContentService.createTextOutput(JSON.stringify(responseObj)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("PesanAnonim") || ss.insertSheet("PesanAnonim"); sheet.appendRow([new Date(), data.pesan]); return ContentService.createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
